\documentclass[border=5pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{arrows.meta, positioning, decorations.pathreplacing}
\usepackage{xcolor}

\definecolor{Garnet}{HTML}{73000A}
\definecolor{NinetyBlack}{HTML}{363636}
\definecolor{SeventyBlack}{HTML}{5C5C5C}
\definecolor{FiftyBlack}{HTML}{A2A2A2}
\definecolor{ThirtyBlack}{HTML}{C7C7C7}
\definecolor{TenBlack}{HTML}{ECECEC}
\definecolor{Atlantic}{HTML}{466A9F}
\definecolor{Horseshoe}{HTML}{65780B}
\definecolor{Sandstorm}{HTML}{FFF2E3}

\tikzset{every node/.append style={rounded corners=0pt}}

\begin{document}
\begin{tikzpicture}[
    lbl/.style={font=\small\bfseries, color=NinetyBlack},
    annot/.style={font=\scriptsize, color=SeventyBlack},
]

\node[font=\bfseries\large, color=Garnet] at (5.5, 6.5) {Normal vs Doppler Echo Byte Layout};

% Bit position labels
\foreach \i/\v in {0/7, 1/6, 2/5, 3/4, 4/3, 5/2, 6/1, 7/0} {
    \node[font=\tiny\bfseries, color=FiftyBlack] at (0.5 + \i*1.0, 5.6) {bit \v};
}

% ── Normal Mode ──
\node[font=\bfseries\color{Garnet}, anchor=east] at (-0.3, 4.8) {Normal Mode:};
\draw[thick] (0, 4.3) rectangle (8, 5.3);
\foreach \x in {1,...,7} {
    \draw[thin, ThirtyBlack] (\x, 4.3) -- (\x, 5.3);
}
\node[font=\small] at (4, 4.8) {Intensity (8 bits, 0--252)};
\draw[decorate, decoration={brace, amplitude=4pt}, thick, Garnet]
    (0, 5.4) -- (8, 5.4) node[midway, above=4pt, font=\scriptsize\bfseries, color=Garnet] {Full 8-bit dynamic range};

% ── Doppler Mode ──
\node[font=\bfseries\color{Garnet}, anchor=east] at (-0.3, 3.0) {Doppler Mode:};
\draw[thick] (0, 2.5) rectangle (8, 3.5);
% D flag bit (bit 7)
\draw[thick, fill=Atlantic!20] (0, 2.5) rectangle (1, 3.5);
\node[font=\small\bfseries] at (0.5, 3.0) {D};
% Intensity bits (6:0)
\foreach \x in {2,...,7} {
    \draw[thin, ThirtyBlack] (\x, 2.5) -- (\x, 3.5);
}
\draw[thick] (1, 2.5) -- (1, 3.5);
\node[font=\small] at (4.5, 3.0) {Intensity (7 bits, 0--126)};

\draw[decorate, decoration={brace, amplitude=4pt, mirror}, thick, Atlantic]
    (0, 2.4) -- (1, 2.4) node[midway, below=4pt, font=\scriptsize\bfseries, color=Atlantic] {D flag};
\draw[decorate, decoration={brace, amplitude=4pt, mirror}, thick, SeventyBlack]
    (1, 2.4) -- (8, 2.4) node[midway, below=4pt, font=\scriptsize, color=SeventyBlack] {7-bit intensity};

% ── Status Field ──
\node[font=\bfseries\color{Garnet}, anchor=east] at (-0.3, 0.3) {Status Field:};

% Draw a wider status field (9+ bits)
\draw[thick, fill=Garnet!12] (0, -0.2) rectangle (1, 0.8);
\node[font=\scriptsize\bfseries] at (0.5, 0.3) {Dop};
\node[font=\tiny, color=FiftyBlack, above=1pt] at (0.5, 0.8) {bit 8};

\draw[thick, fill=ThirtyBlack!15] (1, -0.2) rectangle (5, 0.8);
\node[font=\scriptsize, color=FiftyBlack] at (3, 0.3) {Reserved};
\node[font=\tiny, color=FiftyBlack, above=1pt] at (3, 0.8) {bits 7:3};

\draw[thick, fill=Horseshoe!15] (5, -0.2) rectangle (8, 0.8);
\node[font=\scriptsize\bfseries] at (6.5, 0.3) {Pulse Type};
\node[font=\tiny, color=FiftyBlack, above=1pt] at (6.5, 0.8) {bits 2:0};

% Status value examples
\node[draw, thick, fill=TenBlack, text width=10cm, align=left, font=\scriptsize, inner sep=6pt] at (5, -1.5) {
\textbf{Status values:}
\texttt{0x01} (1) = Normal, long pulse \quad
\texttt{0x04} (4) = Normal, short pulse \\
\texttt{0x101} (257) = Doppler, long pulse \quad
\texttt{0x104} (260) = Doppler, short pulse
};

% Extraction formulas
\node[draw, thick, fill=Sandstorm, text width=10cm, align=left, font=\scriptsize, inner sep=6pt] at (5, -3.0) {
\textbf{Extraction:}
D flag = \texttt{byte \& 0x80} (0 = stationary, 1 = moving target) \\
Intensity = \texttt{byte \& 0x7F} (masked to 7 bits) \\
\textbf{Example:} \texttt{0xC7} $\rightarrow$ D = 1 (moving), intensity = \texttt{0x47} = 71
};

\end{tikzpicture}
\end{document}
